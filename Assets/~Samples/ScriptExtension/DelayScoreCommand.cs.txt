using System;
using Inno.VisualNovel.Model.Extensions;
using Inno.VisualNovel.Model.Values;
using Inno.VisualNovel.Narrative.Extensions;

namespace Inno.VisualNovel.Samples.ScriptExtension;

/// <summary>Demonstrates a restorable command whose complete continuation lives in neutral state.</summary>
public sealed class DelayScoreCommand : NovelCommand
{
    private bool m_disposed;
    /// <inheritdoc />
    public override NovelCommandDefinition definition => new()
    {
        id = "sample.delay-score", title = "Delayed score", help = "Returns an Integer after a logical delay; supports saving and rollback.",
        capabilities = CommandCapabilities.Preview | CommandCapabilities.Save | CommandCapabilities.Rollback | CommandCapabilities.Cancel,
        hasResult = true, resultDefault = NovelValue.Integer(0), parameters = [
            new() { id = "seconds", label = "Wait seconds", defaultValue = NovelValue.Number(1) },
            new() { id = "score", label = "Returned score", defaultValue = NovelValue.Integer(10) }]
    };
    /// <inheritdoc />
    public override bool Execute(
        NovelCommandContext context,
        NovelCommandState state,
        double seconds
    ) {
        ObjectDisposedException.ThrowIf(m_disposed, this);
        double duration = context.Argument("seconds").number;
        if (!double.IsFinite(duration) || duration < 0) throw new InvalidOperationException("Wait seconds must be finite and nonnegative.");
        if (state.elapsed < duration) return false;
        context.Return(context.Argument("score")); return true;
    }
    /// <inheritdoc />
    public override void PrepareRestore(NovelCommandState state)
    {
        ObjectDisposedException.ThrowIf(m_disposed, this);
        if (!state.started || !double.IsFinite(state.elapsed) || state.elapsed < 0) throw new InvalidOperationException("Invalid delayed-command state.");
    }
    /// <inheritdoc />
    public override void Cancel(NovelCommandState state) => m_disposed = true;
    /// <inheritdoc />
    public override void Dispose() => m_disposed = true;
}
